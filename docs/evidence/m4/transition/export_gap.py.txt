"""export_gap.py WAV LOG FPS LABEL: in an exported run's soundtrack, from where the prologue's sound begins: runs below
-65 dB of 5 ms or more, and the deepest 20 ms window against the median of the 0.3 s before it, past the prologue's
first 20 s, where the cut into the game is. Times are from the prologue's start."""
import re, sys
import numpy as np, soundfile as sf
wav, log, fps, label = sys.argv[1], sys.argv[2], float(sys.argv[3]), sys.argv[4]
text = open(log, errors="replace").read()
x, sr = sf.read(wav, always_2d=True)
win = int(0.02 * sr)
rms = np.array([20 * np.log10(np.sqrt(np.mean(x[i:i + win] ** 2)) + 1e-9) for i in range(0, len(x) - win, win // 4)])
t = np.arange(len(rms)) * (win / 4) / sr
start = next(i for i in range(len(rms)) if (rms[i:i + 100] > -50).all())
m = np.abs(x).max(axis=1)
q = np.where(m[int(t[start] * sr):] < 10 ** (-65 / 20))[0]
gaps = []
if len(q):
    for r in np.split(q, np.where(np.diff(q) > 1)[0] + 1):
        if len(r) >= sr * 0.005:
            gaps.append((r[0] / sr, len(r) / sr * 1000))
best = (0.0, 0.0)
# The cut comes 32.9 s after the prologue begins in every run of the export at 60 fps; the dip is looked for round it.
lo, hi = start + int(32.3 / (win / 4 / sr)), start + int(33.6 / (win / 4 / sr))
for i in range(lo, min(hi, len(rms))):
    before = np.median(rms[max(0, i - 60):i])
    if before - rms[i] > best[0]:
        best = (before - rms[i], t[i] - t[start])
finished = "prologue finished" in text
started = "session start" in text
print(f"{label}: game started {started}, prologue finished {finished}; "
      + ("below -65 dB: " + ", ".join(f"{g[1]:.1f} ms at {g[0]:.2f} s" for g in gaps) if gaps else "nothing below -65 dB")
      + f"; deepest 20 ms dip within 32.3-33.6 s: {best[0]:.1f} dB at {best[1]:.2f} s")
