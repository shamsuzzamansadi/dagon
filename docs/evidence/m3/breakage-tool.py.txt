def run(cmd, cwd, env=ENV, timeout=None):
    return subprocess.run(cmd, cwd=cwd, env=env, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, errors="replace", timeout=timeout)

def build(cwd):
    result = run([SCONS, "platform=linux", "target=template_debug", "-j3"], cwd)
    return result.returncode == 0, result.stdout

def test(cwd):
    lines = []
    unit = run([SCONS, "unit_tests"], cwd)
    for line in unit.stdout.splitlines():
        if "ERROR: CHECK" in line or "ERROR: REQUIRE" in line:
            lines.append(line[:170])
        if line.startswith("[doctest] test cases"):
            lines.append(line)
    storage = tempfile.mkdtemp(prefix="dagon-test-", dir=ENV["TMPDIR"])
    env = dict(ENV, DAGON_STORAGE_DIR=storage)
    try:
        started = time.time()
        godot = run([GODOT, "--headless", "--path", "project", "--fixed-fps", "60", "--script", "res://tests/run_tests.gd"], cwd, env=env, timeout=900)
        stdout = godot.stdout
    except subprocess.TimeoutExpired as error:
        stdout = (error.stdout or "") if isinstance(error.stdout, str) else ""
        stdout += "\nTIMED OUT\n"
    finally:
        shutil.rmtree(storage, ignore_errors=True)
    tests = [line for line in stdout.splitlines() if line.startswith("FAIL") or line.startswith("     ") or line.startswith("RESULT") or "TIMED OUT" in line or "Parse Error" in line]
    return lines, tests, unit.returncode

def caught(unit_lines, tests):
    fails = [l for l in tests if l.startswith("FAIL")]
    unit_failed = any("ERROR:" in l for l in unit_lines) or any("| 0 failed" not in l for l in unit_lines if l.startswith("[doctest]"))
    result = [l for l in tests if l.startswith("RESULT")]
    clean = result and result[-1].endswith("failed=0")
    return bool(fails) or unit_failed or not clean

results = {}
def worker(index, jobs, out):
    cwd = f"{ROOT}/w{index}"
    out.write(f"########## worker {index}\n=== control: no change\n"); out.flush()
    ok, log = build(cwd)
    if not ok:
        out.write("CONTROL DID NOT BUILD\n" + log[-2000:]); return
    run(["sh", "tools/prepare_project.sh"], cwd)
    run([GODOT, "--headless", "--path", "project", "--import"], cwd)
    unit, tests, _ = test(cwd)
    out.write("\n".join(unit + [l for l in tests if l.startswith(("FAIL", "RESULT"))]) + "\n\n"); out.flush()
    while True:
        try:
            item = jobs.get_nowait()
        except queue.Empty:
            break
        order, (name, path, old, new) = item
        full = os.path.join(cwd, path)
        text = open(full).read()
        if old not in text:
            out.write(f"=== {name}\nNOT APPLIED: the text was not found in {path}\n\n"); out.flush()
            results[order] = (name, "not applied", 0)
            continue
        open(full, "w").write(text.replace(old, new, 1))
        section = [f"=== {name}"]
        verdict = "caught"
        count = 0
        ok, log = build(cwd) if path.startswith("src") else (True, "")
        if not ok:
            errs = [l for l in log.splitlines() if " error" in l or "error:" in l][:3]
            section += ["DID NOT COMPILE, so this is not a result:"] + [e[:200] for e in errs]
            verdict = "did not compile"
        else:
            unit, tests, _ = test(cwd)
            shown = [l for l in unit if "ERROR" in l][:3] + tests[:14]
            section += [l[:230] for l in shown]
            result = [l for l in tests if l.startswith("RESULT")]
            if result:
                section.append(result[-1])
            count = sum(1 for l in tests if l.startswith("FAIL"))
            if not caught(unit, tests):
                verdict = "NOT CAUGHT"
                section.append("NOT CAUGHT")
        out.write("\n".join(section) + "\n\n"); out.flush()
        results[order] = (name, verdict, count)
        run(["git", "checkout", "--", path], cwd)
        if path.startswith("src"):
            build(cwd)
    out.write(f"=== control after, worker {index}\n")
    unit, tests, _ = test(cwd)
    out.write("\n".join(unit + [l for l in tests if l.startswith(("FAIL", "RESULT"))]) + "\n\n"); out.flush()

def main():
    workers = int(sys.argv[1])
    only = sys.argv[2:]  # optional substrings to run a subset
    # What the first full pass at 2cae7bb did not catch, and the replacements for its two invalid entries.
    gaps = ("A wrong stone leaves the prompts up", "Solved stones keep reacting", "The flee awaits no trigger", "A looping clip is played",
            "The gate changes in a physics callback", "The scene changes inside the finalizer", "Restoring leaves the atmosphere clip running",
            "The lockout runs on while paused", "The flee timer and watchdog run on while paused", "Clips run on while paused", "A polygon crosses itself")
    new_from = next(i for i, b in enumerate(BREAKAGES) if b[0] == "A wrong stone keeps the attempt's progress")
    items = BREAKAGES[:new_from] + [b for b in BREAKAGES[new_from:] if any(g in b[0] for g in gaps)]
    if only:
        items = [b for b in BREAKAGES if any(o in b[0] for o in only)]
    jobs = queue.Queue()
    for pair in enumerate(items):
        jobs.put(pair)
    commit = run(["git", "rev-parse", "--short", "HEAD"], f"{ROOT}/w0").stdout.strip()
    files = [open(f"{ROOT}/out-{i}.txt", "w") for i in range(workers)]
    threads = [threading.Thread(target=worker, args=(i, jobs, files[i])) for i in range(workers)]
    for t in threads: t.start()
    for t in threads: t.join()
    for f in files: f.close()
    with open(f"{ROOT}/summary.txt", "w") as summary:
        tally = {}
        for order in sorted(results):
            name, verdict, count = results[order]
            tally[verdict] = tally.get(verdict, 0) + 1
            summary.write(f"{verdict:16} {count:4} failing  {name}\n")
        summary.write(f"\ncommit {commit}, {len(results)} breakages: {tally}\n")
    print(open(f"{ROOT}/summary.txt").read())

main()
